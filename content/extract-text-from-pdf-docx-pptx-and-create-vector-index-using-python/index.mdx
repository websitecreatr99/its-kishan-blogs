## Function to extract text from PPTX
```python title="pptx" caption="this file contains python code for pptx"
import pptx 
import os

def extract_fromPPT(path_file): 
    """ Extract text from a .pptx file. """ 
    presentation = pptx.Presentation(path_file) 
    text = [] 
    for slide in presentation.slides: 
        for shape in slide.shapes: 
            if hasattr(shape, "text"): 
                text.append(shape.text) 
    return ' '.join(text) 
```
## Function to extract text from DOCX
```python title="docx" caption="this file contains python code for docx"
import docx 
import os

def extract_fromDOC(path_file): 
    """ Extract text from a .docx file. """ 
    doc = docx.Document(path_file) 
    text = [] 
    for paragraph in doc.paragraphs: 
        text.append(paragraph.text) 
    return ' '.join(text)
```

## Function to extract text from PDF
```python title="pdf" caption="this file contains python code for pdf"
import PyPDF2
import os

def extract_fromPDF(path_file):
    pdf_file_obj = open(path_file, "rb") 
    pdf_reader = PdfReader(pdf_file_obj) 
    num_pages = len(pdf_reader.pages) 
    str_extracted = "" 
    for page_num in range(num_pages): 
        page_obj = pdf_reader.pages[page_num] 
        str_extracted += page_obj.extract_text() + '/n/n' 
    pdf_file_obj.close() 
    return str_extracted
```
